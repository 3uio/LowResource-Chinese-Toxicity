#   python -m pip install -r scripts/requirements.txt

# ---------------------------------------------------------------------------
# zhconv —— 【必需】繁转简，归一化流程的第 5 步
# ---------------------------------------------------------------------------
# 为什么必须装：
#   归一化口径（与 Track A 严格一致）共五步：
#       NFKC → 去零宽与BOM → 去空白与标点 → 小写 → 繁转简(zhconv) → SHA-256
#   缺了 zhconv，validate_schema.py 算出的哈希会与 A 不一致，
#   **含繁体字的文本会漏报**，跨表查重等于没查。
#
# 实测证据（用 A 交付的表2 种子预留清单 2,041 条比对）：
#   不装 zhconv（4 步）：2,025/2,041 一致，16 条不一致
#   这 16 条全部含繁体字（來/們/國/後/無/爭/麽/糞/龜/師/婦/慣/內/豬/賴…）
#
# 缺库时 validate_schema.py 会**显式报警**（不静默降级），
# 但报警不等于安全 —— 请务必装上。
zhconv>=1.4.3

# ---------------------------------------------------------------------------
# pypinyin —— 【可选】只影响两个规则
# ---------------------------------------------------------------------------
# 装上以后 PY_Full / PY_Abbr 两个规则才能对任意汉字生效，
# 否则只能对脚本内置的极小占位映射表生效。
pypinyin>=0.51
